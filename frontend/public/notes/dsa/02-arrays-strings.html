02Basic Array Operations

Find the maximum and second maximum

static int secondMax(int[] a) {
    int max = Integer.MIN_VALUE, second = Integer.MIN_VALUE;
    for (int x : a) {
        if (x > max) { second = max; max = x; }
        else if (x > second && x != max) second = x;
    }
    return second;
}

Reverse in place

static void reverse(int[] a, int l, int r) {
    while (l < r) {
        int t = a[l]; a[l] = a[r]; a[r] = t;
        l++; r--;
    }
}

Rotate an array right by k (reversal trick, O(n) time, O(1) space)

static void rotate(int[] a, int k) {
    int n = a.length;
    k %= n;
    reverse(a, 0, n - 1);        // whole array
    reverse(a, 0, k - 1);        // first k
    reverse(a, k, n - 1);        // the rest
}
// [1,2,3,4,5,6,7], k=3 → [5,6,7,1,2,3,4]

Remove duplicates from a sorted array (in place)

static int removeDuplicates(int[] a) {
    if (a.length == 0) return 0;
    int w = 1;                                  // write position
    for (int r = 1; r < a.length; r++)
        if (a[r] != a[r - 1]) a[w++] = a[r];
    return w;                                   // new length
}

Move zeroes to the end (keep order)

static void moveZeroes(int[] a) {
    int w = 0;
    for (int r = 0; r < a.length; r++)
        if (a[r] != 0) { int t = a[w]; a[w] = a[r]; a[r] = t; w++; }
}

Merge two sorted arrays

static int[] merge(int[] a, int[] b) {
    int[] res = new int[a.length + b.length];
    int i = 0, j = 0, k = 0;
    while (i < a.length && j < b.length) res[k++] = a[i] <= b[j] ? a[i++] : b[j++];
    while (i < a.length) res[k++] = a[i++];
    while (j < b.length) res[k++] = b[j++];
    return res;
}

03Two Pointers

Use two indexes that move through the data, often turning O(n²) into O(n).

VariantSetupExample
Opposite endsl = 0, r = n-1, move inwardPair sum in sorted array, palindrome, container with most water
Same direction (fast/slow)Both start leftRemove duplicates, cycle detection, partitioning

Pair sum in a sorted array

static int[] twoSumSorted(int[] a, int target) {
    int l = 0, r = a.length - 1;
    while (l < r) {
        int s = a[l] + a[r];
        if (s == target) return new int[]{l, r};
        if (s < target) l++; else r--;          // sorted order tells us which side to move
    }
    return new int[]{-1, -1};
}

Three sum (all unique triplets that add to 0)

static List<List<Integer>> threeSum(int[] a) {
    Arrays.sort(a);
    List<List<Integer>> res = new ArrayList<>();
    for (int i = 0; i < a.length - 2; i++) {
        if (i > 0 && a[i] == a[i - 1]) continue;          // skip duplicates
        int l = i + 1, r = a.length - 1;
        while (l < r) {
            int s = a[i] + a[l] + a[r];
            if (s == 0) {
                res.add(List.of(a[i], a[l], a[r]));
                while (l < r && a[l] == a[l + 1]) l++;
                while (l < r && a[r] == a[r - 1]) r--;
                l++; r--;
            } else if (s < 0) l++;
            else r--;
        }
    }
    return res;                                           // O(n²) time
}

Container with most water

static int maxArea(int[] h) {
    int l = 0, r = h.length - 1, best = 0;
    while (l < r) {
        best = Math.max(best, Math.min(h[l], h[r]) * (r - l));
        if (h[l] < h[r]) l++; else r--;          // move the shorter wall
    }
    return best;
}

Dutch national flag: sort 0s, 1s and 2s in one pass

static void sortColors(int[] a) {
    int low = 0, mid = 0, high = a.length - 1;
    while (mid <= high) {
        if (a[mid] == 0) swap(a, low++, mid++);
        else if (a[mid] == 1) mid++;
        else swap(a, mid, high--);
    }
}

04Sliding Window

Maintain a window [left, right] over a contiguous range and slide it, updating the answer incrementally instead of recomputing.

Fixed-size window: maximum sum of k consecutive elements

static int maxSumK(int[] a, int k) {
    int sum = 0;
    for (int i = 0; i < k; i++) sum += a[i];
    int best = sum;
    for (int i = k; i < a.length; i++) {
        sum += a[i] - a[i - k];                  // add new element, drop the old one
        best = Math.max(best, sum);
    }
    return best;
}

Variable-size window: longest substring without repeating characters

static int lengthOfLongestSubstring(String s) {
    Map<Character, Integer> last = new HashMap<>();
    int left = 0, best = 0;
    for (int right = 0; right < s.length(); right++) {
        char c = s.charAt(right);
        if (last.containsKey(c) && last.get(c) >= left) left = last.get(c) + 1;
        last.put(c, right);
        best = Math.max(best, right - left + 1);
    }
    return best;                                  // O(n)
}

Variable-size window: smallest subarray with sum ≥ target (positives only)

static int minSubArrayLen(int target, int[] a) {
    int left = 0, sum = 0, best = Integer.MAX_VALUE;
    for (int right = 0; right < a.length; right++) {
        sum += a[right];
        while (sum >= target) {                   // shrink while still valid
            best = Math.min(best, right - left + 1);
            sum -= a[left++];
        }
    }
    return best == Integer.MAX_VALUE ? 0 : best;
}

Template for variable windows

for right in 0..n-1:
    add a[right] to the window state
    while window is invalid (or, for "minimum", while valid):
        remove a[left]; left++
    update the answer using the current window

05Prefix Sums

prefix[i] = sum of the first i elements. Then any range sum is O(1) after O(n) preprocessing.

int[] prefix = new int[n + 1];
for (int i = 0; i < n; i++) prefix[i + 1] = prefix[i] + a[i];

int rangeSum(int l, int r) { return prefix[r + 1] - prefix[l]; }      // sum of a[l..r]
  • 2D prefix sums answer rectangle-sum queries in O(1)
  • Difference arrays do the reverse: add a value to a range in O(1) (diff[l] += v; diff[r+1] -= v), then prefix-sum once to rebuild
  • Combine with a HashMap to count subarrays with a given sum (see the Hashing topic)

Product of array except self (no division)

static int[] productExceptSelf(int[] a) {
    int n = a.length;
    int[] res = new int[n];
    res[0] = 1;
    for (int i = 1; i < n; i++) res[i] = res[i - 1] * a[i - 1];     // product of everything to the left
    int right = 1;
    for (int i = n - 1; i >= 0; i--) { res[i] *= right; right *= a[i]; }   // multiply by everything to the right
    return res;
}

06Kadane's Algorithm: Maximum Subarray Sum

static int maxSubArray(int[] a) {
    int cur = a[0], best = a[0];
    for (int i = 1; i < a.length; i++) {
        cur = Math.max(a[i], cur + a[i]);         // extend the current subarray or start a new one
        best = Math.max(best, cur);
    }
    return best;                                   // O(n) time, O(1) space
}
// [-2,1,-3,4,-1,2,1,-5,4] → 6  (subarray [4,-1,2,1])

07Majority Element: Boyer-Moore Voting

static int majority(int[] a) {                     // element appearing more than n/2 times
    int candidate = 0, count = 0;
    for (int x : a) {
        if (count == 0) candidate = x;
        count += (x == candidate) ? 1 : -1;
    }
    return candidate;                              // O(n), O(1)
}

Best time to buy and sell stock

static int maxProfit(int[] prices) {
    int min = Integer.MAX_VALUE, best = 0;
    for (int p : prices) {
        min = Math.min(min, p);
        best = Math.max(best, p - min);
    }
    return best;
}

Trapping rain water

static int trap(int[] h) {
    int l = 0, r = h.length - 1, leftMax = 0, rightMax = 0, water = 0;
    while (l < r) {
        if (h[l] < h[r]) {
            leftMax = Math.max(leftMax, h[l]);
            water += leftMax - h[l++];
        } else {
            rightMax = Math.max(rightMax, h[r]);
            water += rightMax - h[r--];
        }
    }
    return water;
}

09Matrices (2D Arrays)

Traversal and rotation

for (int i = 0; i < rows; i++)
    for (int j = 0; j < cols; j++) { /* grid[i][j] */ }

// Rotate an n×n matrix 90° clockwise in place: transpose, then reverse each row
static void rotate(int[][] m) {
    int n = m.length;
    for (int i = 0; i < n; i++)
        for (int j = i + 1; j < n; j++) { int t = m[i][j]; m[i][j] = m[j][i]; m[j][i] = t; }
    for (int[] row : m) reverse(row, 0, n - 1);
}

Spiral order

static List<Integer> spiral(int[][] m) {
    List<Integer> res = new ArrayList<>();
    int top = 0, bottom = m.length - 1, left = 0, right = m[0].length - 1;
    while (top <= bottom && left <= right) {
        for (int j = left; j <= right; j++) res.add(m[top][j]);
        top++;
        for (int i = top; i <= bottom; i++) res.add(m[i][right]);
        right--;
        if (top <= bottom) { for (int j = right; j >= left; j--) res.add(m[bottom][j]); bottom--; }
        if (left <= right) { for (int i = bottom; i >= top; i--) res.add(m[i][left]); left++; }
    }
    return res;
}

Search in a row-wise and column-wise sorted matrix: O(m + n)

static boolean search(int[][] m, int target) {
    int r = 0, c = m[0].length - 1;                 // start at top-right
    while (r < m.length && c >= 0) {
        if (m[r][c] == target) return true;
        if (m[r][c] > target) c--; else r++;
    }
    return false;
}

Direction arrays for grids

int[] dr = {-1, 1, 0, 0};
int[] dc = {0, 0, -1, 1};
for (int d = 0; d < 4; d++) {
    int nr = r + dr[d], nc = c + dc[d];
    if (nr < 0 || nc < 0 || nr >= rows || nc >= cols) continue;
}

int[] freq = new int[26];

Valid anagram

static boolean isAnagram(String a, String b) {
    if (a.length() != b.length()) return false;
    int[] f = new int[26];
    for (int i = 0; i < a.length(); i++) { f[a.charAt(i) - 'a']++; f[b.charAt(i) - 'a']--; }
    for (int x : f) if (x != 0) return false;
    return true;
}

Valid palindrome (ignore non-alphanumerics, case-insensitive)

static boolean isPalindrome(String s) {
    int l = 0, r = s.length() - 1;
    while (l < r) {
        while (l < r && !Character.isLetterOrDigit(s.charAt(l))) l++;
        while (l < r && !Character.isLetterOrDigit(s.charAt(r))) r--;
        if (Character.toLowerCase(s.charAt(l++)) != Character.toLowerCase(s.charAt(r--))) return false;
    }
    return true;
}

Longest palindromic substring: expand around centre, O(n²)

static String longestPalindrome(String s) {
    int start = 0, end = 0;
    for (int i = 0; i < s.length(); i++) {
        int len = Math.max(expand(s, i, i), expand(s, i, i + 1));   // odd and even centres
        if (len > end - start + 1) { start = i - (len - 1) / 2; end = i + len / 2; }
    }
    return s.substring(start, end + 1);
}
static int expand(String s, int l, int r) {
    while (l >= 0 && r < s.length() && s.charAt(l) == s.charAt(r)) { l--; r++; }
    return r - l - 1;
}

Reverse words in a string

static String reverseWords(String s) {
    String[] parts = s.trim().split("\\s+");
    StringBuilder sb = new StringBuilder();
    for (int i = parts.length - 1; i >= 0; i--) {
        sb.append(parts[i]);
        if (i > 0) sb.append(' ');
    }
    return sb.toString();
}

Longest common prefix

static String lcp(String[] strs) {
    String prefix = strs[0];
    for (String s : strs)
        while (!s.startsWith(prefix)) prefix = prefix.substring(0, prefix.length() - 1);
    return prefix;
}

String compression (aaabb → a3b2)

static String compress(String s) {
    StringBuilder sb = new StringBuilder();
    for (int i = 0, j; i < s.length(); i = j) {
        j = i;
        while (j < s.length() && s.charAt(j) == s.charAt(i)) j++;
        sb.append(s.charAt(i)).append(j - i);
    }
    return sb.toString();
}

First non-repeating character

static int firstUnique(String s) {
    int[] f = new int[26];
    for (char c : s.toCharArray()) f[c - 'a']++;
    for (int i = 0; i < s.length(); i++) if (f[s.charAt(i) - 'a'] == 1) return i;
    return -1;
}

11String Matching Algorithms

AlgorithmTimeIdea
NaiveO(n·m)Compare at every position

Rabin-KarpO(n + m) averageRolling hash of each window

// KMP: build the LPS array
static int[] lps(String p) {
    int[] lps = new int[p.length()];
    for (int i = 1, len = 0; i < p.length(); ) {
        if (p.charAt(i) == p.charAt(len)) lps[i++] = ++len;
        else if (len > 0) len = lps[len - 1];
        else lps[i++] = 0;
    }
    return lps;
}

static int kmpSearch(String text, String p) {
    int[] lps = lps(p);
    for (int i = 0, j = 0; i < text.length(); ) {
        if (text.charAt(i) == p.charAt(j)) { i++; j++; if (j == p.length()) return i - j; }
        else if (j > 0) j = lps[j - 1];
        else i++;
    }
    return -1;
}

12Edge Cases Checklist

  • Empty array/string, one element, two elements
  • All elements equal; already sorted; reverse sorted
  • Negative numbers, zeros, Integer.MAX_VALUE/MIN_VALUE (overflow)
  • Duplicates; unicode or upper/lower case; leading/trailing spaces
  • k larger than the array length (use k %= n)

13Quick Recap

  • Arrays: O(1) access, O(n) middle insert/delete; strings are immutable, use StringBuilder
  • Two pointers (sorted data, in-place edits) and sliding window (contiguous ranges) cut O(n²) to O(n)
  • Prefix sums answer range queries; Kadane finds the best subarray; Boyer-Moore finds the majority
  • Rotate with three reversals; transpose + reverse rows rotates a matrix
  • Use int[26] for letter counts; know KMP for linear-time pattern search
